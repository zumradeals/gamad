async function who(name) {
  if (users[name]) return users[name];
  const ctx = await browser.newContext({ viewport: vp, hasTouch: vp.width < 600 });
  const p = await ctx.newPage(); p.on('dialog', d => d.accept());
  p._errs = []; p.on('pageerror', e => p._errs.push(String(e).slice(0, 120))); p.on('console', m => { if (m.type() === 'error') p._errs.push(m.text().slice(0, 120)); });
  await p.goto(base + '/login'); await p.fill('input[name=email]', `${name}@recette.test`); await p.fill('input[name=password]', 'recette-2026');
  await Promise.all([p.waitForURL(u => !u.pathname.startsWith('/login'), { timeout: 15000 }), p.click('button[type=submit]')]);
  return (users[name] = p);
}

}
