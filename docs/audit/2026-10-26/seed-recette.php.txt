$mk = function (string $name, string $email, array $off = [], array $want = [], ?string $city = null, $super = false) {
    $f = User::factory(); if ($super) $f = $f->superAdmin();
    $u = $f->create(['name' => $name, 'email' => $email, 'password' => 'recette-2026']);
    $c = $city ? City::where('name', $city)->first() : null;

    return $u->fresh();
};

$mk2 = function (User $owner, string $name, string $status) {

};
