import { chromium } from '/opt/node-tools/node_modules/playwright/index.mjs';
const base='http://localhost:8001';
const b=await chromium.launch({executablePath:'/opt/pw-browsers/chromium'});
const c=await b.newContext({viewport:{width:1440,height:900}}); const p=await c.newPage();
const errs=[]; p.on('pageerror',e=>errs.push(String(e).slice(0,120))); p.on('console',m=>{if(m.type()==='error')errs.push(m.text().slice(0,120))});
await p.goto(base+'/login'); await p.fill('input[name=email]','mariam@recette.test'); await p.fill('input[name=password]','recette-2026');
await Promise.all([p.waitForURL(u=>!u.pathname.startsWith('/login')),p.click('button[type=submit]')]);
await p.goto(base+'/agir/besoin'); await p.fill('#body','Brouillon de test pour vérifier le nettoyage local après publication.'); await p.waitForTimeout(600);
console.log('brouillon local avant publication:', await p.evaluate(()=>Object.keys(localStorage).filter(k=>k.startsWith('gamad:draft'))));
await p.click('button:has-text("Continuer")'); await p.waitForSelector('text=Où ?'); await p.click('button:has-text("Continuer")'); await p.waitForSelector('text=Relisez avant de publier');
await p.click('button:has-text("Publier ma demande")'); await p.waitForURL(/publications\//); await p.waitForTimeout(800);
console.log('brouillon local après publication:', await p.evaluate(()=>Object.keys(localStorage).filter(k=>k.startsWith('gamad:draft'))));
await p.goto(base+'/agir/besoin'); await p.waitForTimeout(600);
console.log('avis « reprendre » visible au retour:', await p.isVisible('text=Vous aviez commencé un texte'));
console.log('erreurs:', errs);
await b.close();
